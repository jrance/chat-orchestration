332:def _sccs(nodes: list[str], edges: list[tuple[str, str]]) -> list[list[str]]:
333:    # Tarjan's algorithm
334:    index = 0
335:    indices: dict[str, int] = {}
336:    lowlink: dict[str, int] = {}
337:    stack: list[str] = []
338:    onstack: set[str] = set()
339:    result: list[list[str]] = []
340:    adj: dict[str, list[str]] = {n: [] for n in nodes}
341:    for u, v in edges:
342:        if u in adj:
343:            adj[u].append(v)
344:
345:    def strongconnect(v: str) -> None:
346:        nonlocal index
347:        indices[v] = index
348:        lowlink[v] = index
349:        index += 1
350:        stack.append(v)
351:        onstack.add(v)
352:        for w in adj.get(v, []):
353:            if w not in indices:
354:                strongconnect(w)
355:                lowlink[v] = min(lowlink[v], lowlink[w])
356:            elif w in onstack:
357:                lowlink[v] = min(lowlink[v], indices[w])
358:        # If v is a root node, pop the stack and output an SCC
359:        if lowlink[v] == indices[v]:
360:            scc: list[str] = []
361:            while True:
362:                w = stack.pop()
363:                onstack.remove(w)
364:                scc.append(w)
365:                if w == v:
366:                    break
367:            result.append(scc)
368:
369:    for n in nodes:
370:        if n not in indices:
371:            strongconnect(n)
372:    return result

375:def _detect_orchestration(cfg: GraphConfig) -> tuple[str, dict[str, Any]]:
376:    """Detect orchestration pattern.
377:
378:    Returns (kind, data) where kind in {"single", "sequential", "handoff"}.
379:    """
380:    agents, aedges = _agent_graph(cfg)
381:    if not agents:
382:        raise GraphCompileError("No agent.codeless nodes found")
383:    if len(agents) == 1:
384:        return "single", {"only": next(iter(agents.values()))}
385:
386:    if not aedges:
387:        raise OrchestrationUnsupportedError(
388:            "Multiple agents without edges are unsupported (connect them or reduce to one agent)"
389:        )
390:
391:    # Group chat detection: any SCC size >= 2 or a self-loop
392:    node_ids = list(agents.keys())
393:    scc_list = _sccs(node_ids, aedges)
394:    # Identify SCCs that represent cycles (size >=2) or a self loop present
395:    cyc_sccs: list[list[str]] = [s for s in scc_list if len(s) >= 2]
396:    if not cyc_sccs and _has_self_loop(aedges):
397:        # Find node(s) with self-loop
398:        loop_nodes = [u for u, v in aedges if u == v]
399:        if loop_nodes:
400:            cyc_sccs = [[loop_nodes[0]]]
401:    if cyc_sccs:
402:        # Use the first detected SCC; order participants by appearance in cfg.nodes for determinism
403:        scc_set = set(cyc_sccs[0])
404:        ordered_participants = [
405:            n.id
406:            for n in cfg.nodes
407:            if isinstance(n, AgentNode) and n.id in scc_set
408:        ]
409:        # Compute degrees inside SCC
410:        indeg_scc: dict[str, int] = {aid: 0 for aid in ordered_participants}
411:        outdeg_scc: dict[str, int] = {aid: 0 for aid in ordered_participants}
412:        internal_edges = [(u, v) for u, v in aedges if u in scc_set and v in scc_set]
413:        for u, v in internal_edges:
414:            outdeg_scc[u] += 1
415:            indeg_scc[v] += 1
416:        # Heuristic: moderator if exactly one node has indeg>=2 and outdeg>=2
417:        # OR label contains 'moderator'
418:        hub_candidates = [
419:            aid
420:            for aid in ordered_participants
421:            if indeg_scc[aid] >= 2 and outdeg_scc[aid] >= 2
422:        ]
423:        labels = {aid: agents[aid].label or "" for aid in ordered_participants}
424:        label_mod = [aid for aid, lab in labels.items() if "moderator" in lab.lower()]
425:        mode = "round_robin"
426:        moderator_id: str | None = None
427:        if len(hub_candidates) == 1:
428:            mode = "moderator"
429:            moderator_id = hub_candidates[0]
430:        if label_mod:
431:            mode = "moderator"
432:            moderator_id = label_mod[0]
433:        return "groupchat", {
434:            "participants": ordered_participants,
435:            "mode": mode,
436:            "moderator_id": moderator_id,
437:        }
438:
439:    # indegree / outdegree and adjacency
440:    indeg: dict[str, int] = {aid: 0 for aid in agents}
441:    outdeg: dict[str, int] = {aid: 0 for aid in agents}
442:    adj: dict[str, list[str]] = {aid: [] for aid in agents}
443:    for u, v in aedges:
444:        outdeg[u] += 1
445:        indeg[v] += 1
446:        adj[u].append(v)
447:
448:    roots = [aid for aid, d in indeg.items() if d == 0]
449:    if len(roots) != 1:
450:        raise OrchestrationUnsupportedError(
451:            "Multiple roots detected; fan-out/concurrency is planned for PR 8"
452:        )
453:
454:    root = roots[0]
455:
456:    # Check for cycles and reachability from root via Kahn's algorithm restricted to reachable set
457:    from collections import deque
458:
459:    reach_indeg = indeg.copy()
460:    q = deque([root])
461:    visited: list[str] = []
462:    seen_set: set[str] = set()
463:    while q:
464:        u = q.popleft()
465:        visited.append(u)
466:        seen_set.add(u)
467:        for v in adj.get(u, []):
468:            reach_indeg[v] -= 1
469:            # Enqueue only when all parents seen in reachable subgraph
470:            if reach_indeg[v] == 0 and v not in seen_set:
471:                q.append(v)
472:
473:    if len(visited) != len(agents):
474:        # Either cycles or disconnected components
475:        # Detect cycle quickly: if any node in reachable set still has indegree > 0
476:        has_cycle = any(d > 0 for d in reach_indeg.values())
477:        if has_cycle:
478:            raise OrchestrationUnsupportedError(
479:                "Agent graph contains cycles; group chat planned for PR 9"
480:            )
481:        raise OrchestrationUnsupportedError(
482:            "Agent graph has disconnected components; single-root, connected graphs only"
483:        )
484:
485:    # Sequential if outdegree <= 1 for all and path covers all agents following unique next
486:    if all(outdeg[aid] <= 1 for aid in agents):
487:        # Construct order by following next pointers from root
488:        order: list[str] = []
489:        cur = root
490:        seen: set[str] = set()
491:        ok = True
492:        while True:
493:            order.append(cur)
494:            seen.add(cur)
495:            outs = adj.get(cur, [])
496:            if not outs:
497:                break
498:            if len(outs) > 1:
499:                ok = False
500:                break
501:            nxt = outs[0]
502:            if nxt in seen:
503:                ok = False
504:                break
505:            cur = nxt
506:        if ok and len(order) == len(agents):
507:            return "sequential", {"order": order}
508:
509:    # Otherwise handoff if at least one node has outdegree > 1
510:    if any(outdeg[aid] > 1 for aid in agents):
511:        return "handoff", {"adjacency": adj, "root": root}
512:
513:    # Fallback unsupported pattern
514:    raise OrchestrationUnsupportedError(
515:        "Unsupported agent graph pattern; requires either a chain or branching handoff"
516:    )

519:def _detect_concurrent(
520:    cfg: GraphConfig,
521:) -> tuple[bool, dict[str, Any]]:
522:    """Detect a simple concurrent fan-out/fan-in stage.
523:
524:    Returns (True, data) if detected where data contains:
525:      - root: str (the fan-out source agent id)
526:      - branches: list[str]
527:      - next: str | None (common join agent if present)
528:    """
529:    # Heuristic opt-in: only consider when meta name hints at concurrency
530:    name_hint = (getattr(getattr(cfg, "meta", None), "name", "") or "").lower()
531:    if "concurrent" not in name_hint:
532:        return False, {}
533:
534:    agents: dict[str, AgentNode] = {n.id: n for n in cfg.nodes if isinstance(n, AgentNode)}
535:    aedges: list[tuple[str, str]] = [
536:        (e.from_, e.to) for e in cfg.edges if e.from_ in agents and e.to in agents
537:    ]
538:    if not agents:
539:        return False, {}
540:    # Build degree/adjacency
541:    indeg: dict[str, int] = {aid: 0 for aid in agents}
542:    outdeg: dict[str, int] = {aid: 0 for aid in agents}
543:    adj: dict[str, list[str]] = {aid: [] for aid in agents}
544:    for u, v in aedges:
545:        outdeg[u] += 1
546:        indeg[v] += 1
547:        adj[u].append(v)
548:
549:    # Single root only
550:    roots = [aid for aid, d in indeg.items() if d == 0]
551:    if len(roots) != 1:
552:        return False, {}
553:    root = roots[0]
554:    # Fan-out at root
555:    if outdeg[root] <= 1:
556:        return False, {}
557:    branches = list(adj[root])
558:    if not branches:
559:        return False, {}
560:    # Case 1: leaves
561:    if all(outdeg[b] == 0 for b in branches):
562:        return True, {"root": root, "branches": branches, "next": None}
563:    # Case 2: common join
564:    if all(outdeg[b] == 1 for b in branches):
565:        nexts = [adj[b][0] for b in branches]
566:        uniq = set(nexts)
567:        if len(uniq) == 1:
568:            return True, {"root": root, "branches": branches, "next": nexts[0]}
569:    return False, {}
