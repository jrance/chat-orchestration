16:def slugify_function_name(base: str, node_id: str | None) -> str:
17:    """Return a safe, <=64-char function name with optional node-id suffix.
18:
19:    - Lowercase
20:    - Hyphens and non [a-z0-9_] -> underscore
21:    - Collapse duplicate underscores and trim
22:    - If node_id provided, append '__' + last 8 hex characters (hyphens removed)
23:    """
24:    s = (base or "").lower().replace("-", "_")
25:    s = _ALLOWED_NAME_RE.sub("_", s)
26:    s = re.sub(r"_+", "_", s).strip("_") or "tool"
27:    if node_id:
28:        short = re.sub(r"[^a-zA-Z0-9]", "", node_id)[-8:] or "bind"
29:        s = f"{s}__{short.lower()}"
30:    # Guard length
31:    return s[:64]
