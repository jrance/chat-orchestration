325:def _to_provider_error(err: Exception) -> ProviderError:
326:    # Try to extract status/code/message from OpenAI SDK errors
327:    status_code: int | None = None
328:    code: str | None = None
329:    message = str(err)
330:    try:
331:        status_code = getattr(err, "status_code", None)
332:        code = getattr(getattr(err, "error", None), "code", None) or getattr(err, "code", None)
333:        message_obj = getattr(err, "message", None)
334:        if message_obj is not None:
335:            message = str(message_obj)
336:    except Exception:  # pragma: no cover - best-effort
337:        message = str(err)
338:    return ProviderError(message=message, status_code=status_code, code=code)
